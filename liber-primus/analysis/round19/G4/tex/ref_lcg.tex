% Round 19 / G4 — reference vectors for the `lcg' package (Erich Janka).
% Emits, per seed:
%   LCG <seed> RAW  z1 ... zN     (raw \r@nd state stream, no rejection, no mapping)
%   LCG <seed> R029 v1 ... vN     (\rand with first=0,last=28)
%   LCG <seed> R129 v1 ... vN     (\rand with first=1,last=29)
\documentclass{article}
\usepackage[first=0,last=28,seed=1,quiet]{lcg}
\makeatletter
\newwrite\g@out
\newcount\g@i
\def\NDRAW{64}
\def\seedlist{1,2,3,29,3301,16383,16384,65535,100000,1000000,2000000,2147483646,%
1033,761,845145127,1595277641,1386342,2013,4026}
% raw state stream: drive \r@nd directly
\def\emitraw#1{%
  \cr@nd=#1\relax
  \edef\g@line{}%
  \g@i=0
  \loop\ifnum\g@i<\NDRAW\relax
    \r@nd%
    \edef\g@line{\g@line\space\the\cr@nd}%
    \advance\g@i by 1
  \repeat
  \immediate\write\g@out{LCG #1 RAW\g@line}%
}
\def\emitmapped#1#2#3#4{% seed, first, last, label
  \reinitrand[first=#2,last=#3,seed=#1,quiet]%
  \edef\g@line{}%
  \g@i=0
  \loop\ifnum\g@i<\NDRAW\relax
    \rand%
    \edef\g@line{\g@line\space\therand}%
    \advance\g@i by 1
  \repeat
  \immediate\write\g@out{LCG #1 #4\g@line}%
}
\begin{document}
\immediate\openout\g@out=ref_lcg.txt
\immediate\write\g@out{LCGVERSION 1.3}
\@for\g@s:=\seedlist\do{%
  \expandafter\emitraw\expandafter{\g@s}%
  \expandafter\emitmapped\expandafter{\g@s}{0}{28}{R029}%
  \expandafter\emitmapped\expandafter{\g@s}{1}{29}{R129}%
}
\immediate\closeout\g@out
\makeatother
done
\end{document}
